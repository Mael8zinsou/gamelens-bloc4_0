4. Calcul analytique distribue : fenetre glissante et classement

  --- SQL genere par Snowpark et envoye a l'entrepot ---
  SELECT 
       * 
   FROM (
   SELECT 
      "JEU", 
      "GENRE", 
      "JOUR", 
      "JOUEURS_MOYENS", 
      "MOYENNE_GLISSANTE_7J", 
      rank() OVER (PARTITION BY "GENRE"  ORDER BY 
      "JOUEURS_MOYENS" DESC NULLS LAST ) AS "RANG_DANS_LE_GENRE", 
      round(((100 * "JOUEURS_MOYENS") / sum("JOUEURS_MOYENS") OVER (PARTITION BY "GENRE"  )), 1) AS "PART_DU_GENRE_PCT"
   FROM (
   SELECT 
       * 
   FROM (
   SELECT 
      "JEU", 
      "GENRE", 
      "JOUR", 
      "JOUEURS_MOYENS", 
      "GAME_ID", 
      round(avg("JOUEURS_MOYENS") OVER (PARTITION BY "GAME_ID"  ORDER BY 
      "JOUR" ASC NULLS FIRST  ROWS BETWEEN 6 PRECEDING  AND CURRENT ROW ), 2) AS "MOYENNE_GLISSANTE_7J"
   FROM (
   SELECT 
      "UNIFIED_NAME" AS "JEU", 
      "GENRE", 
      "DAY" AS "JOUR", 
      "AVG_PLAYER_COUNT" AS "JOUEURS_MOYENS", 
      "l_0014_GAME_ID" AS "GAME_ID"
   FROM (
   SELECT  * 
   FROM (
  (
   SELECT 
      "GAME_ID" AS "l_0014_GAME_ID", 
      "DAY", 
      "AVG_PLAYER_COUNT", 
      "MAX_PLAYER_COUNT", 
      "AVG_VIEWER_COUNT", 
      "MAX_VIEWER_COUNT"
   FROM mart.fact_popularity_history
  ) AS SNOWPARK_LEFT 
  INNER JOIN 
  (
   SELECT 
      "GAME_ID" AS "r_0015_GAME_ID", 
      "UNIFIED_NAME", 
      "GENRE"
   FROM mart.dim_games
  ) AS SNOWPARK_RIGHT
   ON ("l_0014_GAME_ID" = "r_0015_GAME_ID")
  )
  )
  )
  )
   WHERE 
      ("JOUR" = DATE '2026-09-01')
  )
  )
   ORDER BY 
      "GENRE" ASC NULLS FIRST, 
      "RANG_DANS_LE_GENRE" ASC NULLS FIRST
  --- fin du SQL ---
