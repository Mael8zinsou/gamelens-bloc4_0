# Outillage Snowflake : dbt et Snowpark.
#
# Ces dependances sont volontairement SEPAREES de requirements.txt et ne sont
# jamais installees dans l'environnement Python du poste. Raison constatee le
# 20/08/2026 : snowflake-snowpark-python impose snowflake-connector-python 4.x,
# incompatible avec la contrainte ~=3.0 de dbt-snowflake 1.8, et remonte au
# passage la version de requests epinglee par requirements.txt.
#
# Elles vivent donc dans leur propre conteneur (docker/snowflake/Dockerfile),
# pour la meme raison qu'Airflow : un jeu de dependances qui entre en conflit
# avec le reste s'isole plutot qu'il ne s'arbitre.
#
# Versions resolues conjointement, sans conflit, et figees ici.
snowflake-snowpark-python==1.54.0
# L'extra pandas apporte pyarrow, requis par write_pandas pour televerser un
# jeu de donnees vers Snowflake sans passer par un fichier intermediaire.
snowflake-connector-python[pandas]==4.7.2
# pandas 2.3.3 et non 2.2.3 comme dans requirements.txt : l'extra pandas du
# connecteur exige une version plus recente, et un pin trop bas rendait la
# resolution impossible. Les deux environnements etant isoles, la divergence
# est sans consequence.
pandas==2.3.3
# Lecture de la couche Silver speed, source de la promotion vers Gold.
psycopg2-binary==2.9.12
# 1.2 minimum, exige par dbt-core 1.12 : le pin a 1.0.1 recopie de
# requirements.txt rendait la resolution impossible.
python-dotenv>=1.2
dbt-core==1.12.2
dbt-snowflake==1.12.0
